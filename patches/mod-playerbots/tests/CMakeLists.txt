    message(FATAL_ERROR "Set PLAYERBOTS_SOURCE to the patched module checkout")

# Compile the real functions with fake game boundaries, never a copied policy.
function(extract_method source signature output)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}")
    file(READ "${source}" text)
    string(FIND "${text}" "${signature}" begin)
    if(begin LESS 0)
        message(FATAL_ERROR "Missing production method: ${signature}")
    endif()
    string(SUBSTRING "${text}" ${begin} -1 tail)
    string(FIND "${tail}" "\n}" end)
    if(end LESS 0)
        message(FATAL_ERROR "Unterminated production method: ${signature}")
    endif()
    math(EXPR size "${end}+2")
    string(SUBSTRING "${tail}" 0 ${size} method)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}.inc" "${method}\n")
endfunction()
set(icc "${PLAYERBOTS_SOURCE}/src/Ai/Raid/ICC")
extract_method("${icc}/ICCTriggers.cpp" "bool IccGunshipRocketPackSetupTrigger::IsActive()" Trigger)
extract_method("${icc}/Action/ICCActions_GSB.cpp" "bool IccGunshipRocketPackSetupAction::Execute(" Execute)
extract_method("${icc}/Action/ICCActions_GSB.cpp" "Item* IccGunshipRocketPackSetupAction::FindRocketPack() const" Find)
extract_method("${icc}/Action/ICCActions_GSB.cpp" "bool IccGunshipRocketPackSetupAction::EquipRocketPack()" Equip)
add_executable(rocket_pack_test RocketPack.cpp)
target_include_directories(rocket_pack_test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

add_test(NAME rocket_pack_test COMMAND rocket_pack_test)
